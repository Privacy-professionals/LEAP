const STORE_KEY = "pp-lang";

class Component extends DCLogic {
  constructor(props) {
    super(props);
    let saved = null;
    try { saved = localStorage.getItem(STORE_KEY); } catch (e) {}
    const fallback = this.props.defaultLang === "en" ? "en" : "ar";
    this.state = {
      lang: saved === "ar" || saved === "en" ? saved : fallback,
      narrow: typeof window !== "undefined" ? window.matchMedia("(max-width:900px)").matches : false,
      navOpen: false,
      scrolled: false,
      active: ""
    };
  }

  componentDidMount() {
    this.applyDoc();
    this.onScroll = () => {
      const s = window.scrollY > 8;
      if (s !== this.state.scrolled) this.setState({ scrolled: s });
    };
    this.onResize = () => {
      const n = window.matchMedia("(max-width:900px)").matches;
      if (n !== this.state.narrow) this.setState({ narrow: n, navOpen: false });
    };
    this.onKey = (e) => { if (e.key === "Escape") this.setState({ navOpen: false }); };
    window.addEventListener("scroll", this.onScroll, { passive: true });
    window.addEventListener("resize", this.onResize);
    document.addEventListener("keydown", this.onKey);
    this.onScroll();
    this.setupReveal();
    this.setupActive();
  }

  componentWillUnmount() {
    window.removeEventListener("scroll", this.onScroll);
    window.removeEventListener("resize", this.onResize);
    document.removeEventListener("keydown", this.onKey);
    if (this.revealSweep) {
      window.removeEventListener("scroll", this.revealSweep);
      window.removeEventListener("load", this.revealSweep);
    }
    if (this.io) this.io.disconnect();
    if (this.io2) this.io2.disconnect();
  }

  componentDidUpdate(prev) {
    if (prev && prev.lang !== this.state.lang) this.applyDoc();
  }

  applyDoc() {
    const lang = this.state.lang;
    const el = document.documentElement;
    el.setAttribute("lang", lang);
    el.setAttribute("dir", lang === "ar" ? "rtl" : "ltr");
    document.title = DICT[lang].pageTitle;
    try { localStorage.setItem(STORE_KEY, lang); } catch (e) {}
  }

  setupReveal() {
    const nodes = Array.prototype.slice.call(document.querySelectorAll("[data-reveal]"));
    const reduce = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
    if (this.props.enableReveal === false || reduce || !("IntersectionObserver" in window)) return;
    nodes.forEach((n) => {
      n.style.opacity = "0";
      n.style.transform = "translateY(20px)";
      n.style.transition = "opacity .7s ease, transform .7s ease";
    });
    const show = (el) => {
      el.style.opacity = "1";
      el.style.transform = "none";
      if (this.io) this.io.unobserve(el);
    };
    this.io = new IntersectionObserver((entries) => {
      entries.forEach((en) => {
        if (en.isIntersecting || en.boundingClientRect.bottom < 0) show(en.target);
      });
    }, { threshold: 0.1, rootMargin: "0px 0px -40px 0px" });
    nodes.forEach((n) => this.io.observe(n));
    this.revealSweep = () => {
      nodes.forEach((n) => {
        if (n.style.opacity !== "1" && n.getBoundingClientRect().top < window.innerHeight) show(n);
      });
    };
    window.addEventListener("scroll", this.revealSweep, { passive: true });
    window.addEventListener("load", this.revealSweep);
    this.revealSweep();
  }

  setupActive() {
    if (!("IntersectionObserver" in window)) return;
    const ids = ["about", "services", "why", "contact"];
    this.io2 = new IntersectionObserver((entries) => {
      entries.forEach((en) => {
        if (en.isIntersecting && en.target.id !== this.state.active) this.setState({ active: en.target.id });
      });
    }, { rootMargin: "-45% 0px -50% 0px" });
    ids.forEach((id) => {
      const el = document.getElementById(id);
      if (el) this.io2.observe(el);
    });
  }

  navLink(id) {
    const on = this.state.active === id;
    return {
      fontSize: "15.5px",
      fontWeight: 500,
      color: on ? "#08663E" : "#4A544F",
      paddingBlock: "6px",
      borderBottom: on ? "2px solid #08663E" : "2px solid transparent",
      transition: "color .2s ease, border-color .2s ease",
      whiteSpace: "nowrap"
    };
  }

  renderVals() {
    const lang = this.state.lang;
    const t = DICT[lang] || DICT.ar;
    const narrow = this.state.narrow;
    const scrolled = this.state.scrolled;
    return {
      t: t,
      siteUrl: this.props.siteUrl || "https://ppros.com.sa/",
      arrowStyle: { transform: lang === "ar" ? "scaleX(-1)" : "none", transition: "transform .22s ease" },
      headerStyle: {
        position: "sticky", top: 0, zIndex: 200,
        background: scrolled ? "rgba(252,253,252,.9)" : "rgba(252,253,252,.55)",
        backdropFilter: "saturate(150%) blur(12px)",
        WebkitBackdropFilter: "saturate(150%) blur(12px)",
        borderBottom: scrolled ? "1px solid #DDE7E1" : "1px solid transparent",
        boxShadow: scrolled ? "0 1px 16px rgba(3,45,32,.06)" : "none",
        transition: "background .25s ease, box-shadow .25s ease, border-color .25s ease"
      },
      navDeskStyle: { display: narrow ? "none" : "flex", alignItems: "center" },
      navA: this.navLink("about"),
      navS: this.navLink("services"),
      navM: this.navLink("contact"),
      navW: this.navLink("why"),
      headerCtaStyle: {
        display: narrow ? "none" : "inline-flex",
        alignItems: "center", justifyContent: "center",
        background: "#08663E", color: "#fff", borderRadius: "999px",
        padding: "10px 20px", fontSize: "14.5px", fontWeight: 600,
        whiteSpace: "nowrap", transition: "background .22s ease"
      },
      burgerStyle: {
        display: narrow ? "inline-flex" : "none",
        alignItems: "center", justifyContent: "center",
        width: "42px", height: "42px", border: "1px solid #DDE7E1",
        borderRadius: "11px", color: "#303533", background: "#FCFDFC"
      },
      mobileMenuStyle: {
        display: narrow && this.state.navOpen ? "block" : "none",
        borderTop: "1px solid #DDE7E1",
        background: "#FCFDFC",
        padding: "6px clamp(18px,4vw,40px) 22px",
        boxShadow: "0 12px 26px rgba(3,45,32,.08)"
      },
      trustStyle: {
        display: this.props.showTrustStrip === false ? "none" : "grid",
        gridTemplateColumns: "repeat(auto-fit,minmax(210px,1fr))",
        gap: "12px 22px",
        marginTop: "34px",
        paddingTop: "26px",
        borderTop: "1px solid #DDE7E1"
      },
      navOpenStr: this.state.navOpen ? "true" : "false",
      toggleLang: () => this.setState({ lang: this.state.lang === "ar" ? "en" : "ar", navOpen: false }),
      toggleNav: () => this.setState({ navOpen: !this.state.navOpen }),
      closeNav: () => this.setState({ navOpen: false })
    };
  }
}
</script>
